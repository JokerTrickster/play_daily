# Task 006: Profile Privacy Toggle UI

## Description

Extend the existing ProfileScreen with room privacy controls, enabling users to toggle between public and private modes. When switching to private mode, display the auto-generated 4-digit password with copy functionality. Implement confirmation dialogs for privacy changes and success notifications showing the new password.

**핵심 목표:**
- Radio button toggle (Public / Private)
- Password display with copy button (private mode only)
- Confirmation dialogs for privacy changes
- Success dialog showing new password

This UI connects to the Privacy Toggle API (Task 003) and provides the primary interface for room owners to manage visibility settings.

## Acceptance Criteria

- [ ] Privacy toggle section added to ProfileScreen (below room name)
- [ ] Radio buttons for Public/Private selection with clear labels
- [ ] Password TextField appears only when Private is selected
- [ ] Copy button next to password with visual feedback (toast/snackbar)
- [ ] Confirmation dialog appears before privacy change with impact explanation
- [ ] Success dialog shows new password when toggling to Private
- [ ] Loading state during API call (disable buttons)
- [ ] Error handling with user-friendly messages
- [ ] UI adapts to existing Material Design theme

@Composable
fun ProfileScreen(viewModel: ProfileViewModel = hiltViewModel()) {
    val roomPrivacy by viewModel.roomPrivacy.collectAsState()
    val isLoading by viewModel.isLoading.collectAsState()

    // Existing profile content...

    // Add Privacy Section
    Card(modifier = Modifier.padding(16.dp)) {
        Column(modifier = Modifier.padding(16.dp)) {
            Text("Room Privacy", style = MaterialTheme.typography.titleMedium)
            Spacer(modifier = Modifier.height(8.dp))

            // Radio buttons
            Row(verticalAlignment = Alignment.CenterVertically) {
                RadioButton(
                    selected = roomPrivacy.isPublic,
                    onClick = { viewModel.showPrivacyConfirmation(true) },
                    enabled = !isLoading
                )
                Text("Public (Discoverable)")
                Spacer(modifier = Modifier.width(16.dp))
                RadioButton(
                    selected = !roomPrivacy.isPublic,
                    onClick = { viewModel.showPrivacyConfirmation(false) },
                    enabled = !isLoading
                )
                Text("Private (Password)")
            }

            // Password field (only if private)
            if (!roomPrivacy.isPublic && roomPrivacy.password != null) {
                Spacer(modifier = Modifier.height(12.dp))
                OutlinedTextField(
                    value = roomPrivacy.password,
                    onValueChange = {},
                    label = { Text("Room Password") },
                    readOnly = true,
                    trailingIcon = {
                        IconButton(onClick = { viewModel.copyPassword() }) {
                            Icon(Icons.Default.ContentCopy, "Copy")
                        }
                    },
                    modifier = Modifier.fillMaxWidth()
                )
                Text(
                    "Share this password for others to join",
                    style = MaterialTheme.typography.bodySmall,
                    color = MaterialTheme.colorScheme.onSurfaceVariant
                )
            }
        }
    }

    // Confirmation Dialog
    if (viewModel.showConfirmationDialog.value) {
        AlertDialog(
            onDismissRequest = { viewModel.dismissConfirmation() },
            title = { Text("Change Room Privacy?") },
            text = {
                Text(if (viewModel.pendingPrivacy.value) {
                    "Your room will be visible in the discovery feed. Anyone can join without a password."
                } else {
                    "Your room will be hidden from discovery. A 4-digit password will be generated for access."
                })
            },
            confirmButton = {
                TextButton(onClick = { viewModel.confirmPrivacyChange() }) {
                    Text("Confirm")
                }
            },
            dismissButton = {
                TextButton(onClick = { viewModel.dismissConfirmation() }) {
                    Text("Cancel")
                }
            }
        )
    }

    // Success Dialog (new password)
    if (viewModel.showSuccessDialog.value) {
        AlertDialog(
            onDismissRequest = { viewModel.dismissSuccess() },
            title = { Text("Privacy Updated") },
            text = {
                Column {
                    Text("Your room is now private.")
                    if (viewModel.newPassword.value != null) {
                        Spacer(modifier = Modifier.height(8.dp))
                        Text(
                            "New Password: ${viewModel.newPassword.value}",
                            style = MaterialTheme.typography.titleLarge,
                            fontWeight = FontWeight.Bold
                        )
                        Spacer(modifier = Modifier.height(4.dp))
                        Text(
                            "Make sure to save this password!",
                            style = MaterialTheme.typography.bodySmall,
                            color = MaterialTheme.colorScheme.error
                        )
                    }
                }
            },
            confirmButton = {
                TextButton(onClick = {
                    viewModel.copyPassword()
                    viewModel.dismissSuccess()
                }) {
                    Text("Copy & Close")
                }
            }
        )
    }
}

### ViewModel Extension

**ProfileViewModel** (`ProfileViewModel.kt`):
```kotlin
@HiltViewModel
class ProfileViewModel @Inject constructor(
    private val roomRepository: RoomRepository,
    private val clipboardManager: ClipboardManager
) : ViewModel() {

    private val _roomPrivacy = MutableStateFlow(RoomPrivacy(isPublic = false, password = null))
    val roomPrivacy: StateFlow<RoomPrivacy> = _roomPrivacy.asStateFlow()

    private val _isLoading = MutableStateFlow(false)
    val isLoading: StateFlow<Boolean> = _isLoading.asStateFlow()

    val showConfirmationDialog = mutableStateOf(false)
    val showSuccessDialog = mutableStateOf(false)
    val pendingPrivacy = mutableStateOf(false)
    val newPassword = mutableStateOf<String?>(null)

    init {
        loadRoomPrivacy()
    }

    private fun loadRoomPrivacy() {
        viewModelScope.launch {
            roomRepository.getCurrentRoomPrivacy()
                .onSuccess { _roomPrivacy.value = it }
                .onFailure { /* handle error */ }
        }
    }

    fun showPrivacyConfirmation(isPublic: Boolean) {
        if (_roomPrivacy.value.isPublic != isPublic) {
            pendingPrivacy.value = isPublic
            showConfirmationDialog.value = true
        }
    }

    fun confirmPrivacyChange() {
        showConfirmationDialog.value = false
        _isLoading.value = true

        viewModelScope.launch {
            roomRepository.updateRoomPrivacy(
                roomId = getCurrentRoomId(),
                isPublic = pendingPrivacy.value
            ).onSuccess { updatedRoom ->
                _roomPrivacy.value = RoomPrivacy(
                    isPublic = updatedRoom.isPublic,
                    password = updatedRoom.password
                )

                if (!updatedRoom.isPublic && updatedRoom.password != null) {
                    newPassword.value = updatedRoom.password
                    showSuccessDialog.value = true
                }
            }.onFailure { error ->
                // Show error snackbar
            }

            _isLoading.value = false
        }
    }

    fun copyPassword() {
        _roomPrivacy.value.password?.let { password ->
            clipboardManager.setPrimaryClip(
                ClipData.newPlainText("Room Password", password)
            )
            // Show toast: "Password copied to clipboard"
        }
    }

    fun dismissConfirmation() {
        showConfirmationDialog.value = false
    }

    fun dismissSuccess() {
        showSuccessDialog.value = false
        newPassword.value = null
    }
}

data class RoomPrivacy(
    val isPublic: Boolean,
    val password: String?
)
```

// Add to existing RoomRepository.kt
suspend fun getCurrentRoomPrivacy(): Result<RoomPrivacy> {
    return try {
        val response = api.getRoomDetails(userPrefs.getCurrentRoomId())
        Result.success(RoomPrivacy(
            isPublic = response.isPublic,
            password = response.password
        ))
    } catch (e: Exception) {
        Result.failure(e)
    }
}

suspend fun updateRoomPrivacy(roomId: Long, isPublic: Boolean): Result<Room> {
    return try {
        val response = api.updateRoomPrivacy(roomId, UpdatePrivacyRequest(isPublic))
        Result.success(response)
    } catch (e: Exception) {
        Result.failure(e)
    }
}
