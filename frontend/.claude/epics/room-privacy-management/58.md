# Task 008: Join Room Screen

## Description

Create a new JoinRoomScreen that allows users to join rooms by entering a room code. When the room is private, display a password modal with a 4-digit PIN input interface. Provide a direct link to the Discover Rooms screen for users who prefer browsing public rooms. Implement comprehensive error handling for invalid codes, wrong passwords, and network failures.

**핵심 목표:**
- Room code text input
- Password modal (4-digit PIN input)
- Link to Discover Rooms screen
- Error handling (invalid code, wrong password)

This screen connects to the Join Room API (Task 005) and provides the direct joining mechanism for users with room codes.

@Composable
fun JoinRoomScreen(
    viewModel: JoinRoomViewModel = hiltViewModel(),

                label = { Text("Room Code") },

                maxLines = 1,
                isError = uiState.error != null,
                supportingText = {
                    if (uiState.error != null) {
                        Text(
                            text = uiState.error,
                            color = MaterialTheme.colorScheme.error
                        )
                    }
                },
                modifier = Modifier.fillMaxWidth(),
                enabled = !uiState.isLoading
            )

            Spacer(modifier = Modifier.height(16.dp))

                onClick = { viewModel.joinByCode() },
                modifier = Modifier.fillMaxWidth(),
                enabled = roomCode.length >= 8 && !uiState.isLoading
            ) {
                if (uiState.isLoading) {
                    CircularProgressIndicator(
                        modifier = Modifier.size(20.dp),
                        color = MaterialTheme.colorScheme.onPrimary
                    )
                    Spacer(modifier = Modifier.width(8.dp))
                }
                Text("Join Room")
            }

            Spacer(modifier = Modifier.height(24.dp))

            // Divider with "OR"
            Row(
                modifier = Modifier.fillMaxWidth(),
                verticalAlignment = Alignment.CenterVertically
            ) {
                HorizontalDivider(modifier = Modifier.weight(1f))
                Text(
                    text = "OR",
                    modifier = Modifier.padding(horizontal = 16.dp),
                    style = MaterialTheme.typography.bodyMedium,
                    color = MaterialTheme.colorScheme.onSurfaceVariant
                )
                HorizontalDivider(modifier = Modifier.weight(1f))
            }

            Spacer(modifier = Modifier.height(24.dp))

            // Discover rooms button
            OutlinedButton(
                onClick = onNavigateToDiscover,
                modifier = Modifier.fillMaxWidth()
            ) {
                Icon(Icons.Default.Explore, contentDescription = null)
                Spacer(modifier = Modifier.width(8.dp))
                Text("Discover Public Rooms")
            }
        }
    }

    // Password modal
    if (uiState.showPasswordModal) {
        PasswordInputDialog(
            onDismiss = { viewModel.cancelPasswordInput() },
            onConfirm = { password -> viewModel.submitPassword(password) },
            isLoading = uiState.isPasswordLoading,
            error = uiState.passwordError
        )
    }

    // Handle successful join
    LaunchedEffect(uiState.joinedRoomId) {
        uiState.joinedRoomId?.let { roomId ->
            onNavigateToRoom(roomId)
            viewModel.clearJoinedRoom()
        }
    }
}

@Composable
fun PasswordInputDialog(
    onDismiss: () -> Unit,
    onConfirm: (String) -> Unit,
    isLoading: Boolean = false,
    error: String? = null
) {
    var password by remember { mutableStateOf("") }

    AlertDialog(
        onDismissRequest = onDismiss,
        title = { Text("Room Password Required") },
        text = {
            Column {
                Text("This room is private. Please enter the 4-digit password.")

                Spacer(modifier = Modifier.height(16.dp))

                // PIN input field
                OutlinedTextField(
                    value = password,
                    onValueChange = { if (it.length <= 4 && it.all { c -> c.isDigit() }) password = it },
                    label = { Text("Password") },
                    keyboardOptions = KeyboardOptions(
                        keyboardType = KeyboardType.NumberPassword,
                        imeAction = ImeAction.Done
                    ),
                    keyboardActions = KeyboardActions(
                        onDone = { if (password.length == 4) onConfirm(password) }
                    ),
                    visualTransformation = PasswordVisualTransformation(),
                    maxLines = 1,
                    isError = error != null,
                    supportingText = {
                        if (error != null) {
                            Text(
                                text = error,
                                color = MaterialTheme.colorScheme.error
                            )
                        }
                    },
                    modifier = Modifier.fillMaxWidth(),
                    enabled = !isLoading
                )

                // Visual dots indicator (alternative to password field)
                Row(
                    modifier = Modifier.fillMaxWidth(),
                    horizontalArrangement = Arrangement.SpaceEvenly
                ) {
                    repeat(4) { index ->
                        Box(
                            modifier = Modifier
                                .size(48.dp)
                                .background(
                                    color = if (index < password.length)
                                        MaterialTheme.colorScheme.primary
                                    else
                                        MaterialTheme.colorScheme.surfaceVariant,
                                    shape = CircleShape
                                ),
                            contentAlignment = Alignment.Center
                        ) {
                            if (index < password.length) {
                                Icon(
                                    Icons.Default.Circle,
                                    contentDescription = null,
                                    modifier = Modifier.size(16.dp),
                                    tint = MaterialTheme.colorScheme.onPrimary
                                )
                            }
                        }
                    }
                }
            }
        },
        confirmButton = {
            TextButton(
                onClick = { onConfirm(password) },
                enabled = password.length == 4 && !isLoading
            ) {
                if (isLoading) {
                    CircularProgressIndicator(modifier = Modifier.size(16.dp))
                    Spacer(modifier = Modifier.width(8.dp))
                }
                Text("Join")
            }
        },
        dismissButton = {
            TextButton(
                onClick = onDismiss,
                enabled = !isLoading
            ) {
                Text("Cancel")
            }
        }
    )
}

### ViewModel Implementation

**JoinRoomViewModel** (`JoinRoomViewModel.kt`):
```kotlin
@HiltViewModel
class JoinRoomViewModel @Inject constructor(
    private val roomRepository: RoomRepository
) : ViewModel() {

    private val _uiState = MutableStateFlow(JoinRoomUiState())
    val uiState: StateFlow<JoinRoomUiState> = _uiState.asStateFlow()

    private val _roomCode = MutableStateFlow("")
    val roomCode: StateFlow<String> = _roomCode.asStateFlow()

    private var pendingRoomCode: String? = null

    fun updateRoomCode(code: String) {
        _roomCode.value = code.uppercase().take(8)
        // Clear error when user types
        if (_uiState.value.error != null) {
            _uiState.value = _uiState.value.copy(error = null)
        }
    }

    fun joinByCode() {
        val code = _roomCode.value.trim()
        if (code.length < 8) {
            _uiState.value = _uiState.value.copy(error = "Room code must be 8 characters")
            return
        }

        _uiState.value = _uiState.value.copy(isLoading = true, error = null)
        pendingRoomCode = code

        viewModelScope.launch {
            roomRepository.joinRoom(roomCode = code, password = null)
                .onSuccess { response ->
                    handleJoinSuccess(response)
                }
                .onFailure { error ->
                    handleJoinError(error)
                }
        }
    }

    fun submitPassword(password: String) {
        val code = pendingRoomCode ?: return

        _uiState.value = _uiState.value.copy(isPasswordLoading = true, passwordError = null)

        viewModelScope.launch {
            roomRepository.joinRoom(roomCode = code, password = password)
                .onSuccess { response ->
                    _uiState.value = _uiState.value.copy(
                        showPasswordModal = false,
                        isPasswordLoading = false
                    )
                    handleJoinSuccess(response)
                }
                .onFailure { error ->
                    when (error) {
                        is PasswordIncorrectException -> {
                            _uiState.value = _uiState.value.copy(
                                isPasswordLoading = false,
                                passwordError = "Incorrect password. Please try again."
                            )
                        }
                        else -> {
                            _uiState.value = _uiState.value.copy(
                                showPasswordModal = false,
                                isPasswordLoading = false
                            )
                            handleJoinError(error)
                        }
                    }
                }
        }
    }

    private fun handleJoinSuccess(response: JoinRoomResponse) {
        _uiState.value = _uiState.value.copy(
            isLoading = false,
            joinedRoomId = response.roomId
        )
    }

    private fun handleJoinError(error: Throwable) {
        val errorMessage = when (error) {
            is RoomNotFoundException -> "Room not found. Please check the code."
            is PasswordRequiredException -> {
                // Show password modal
                _uiState.value = _uiState.value.copy(
                    isLoading = false,
                    showPasswordModal = true
                )
                return
            }
            is RoomFullException -> "Room is full. Cannot join at this time."
            is NetworkException -> "Network error. Please check your connection."
            else -> "Failed to join room. ${error.message}"
        }

        _uiState.value = _uiState.value.copy(
            isLoading = false,
            error = errorMessage
        )
    }

    fun cancelPasswordInput() {
        _uiState.value = _uiState.value.copy(
            showPasswordModal = false,
            isPasswordLoading = false,
            passwordError = null,
            isLoading = false
        )
        pendingRoomCode = null
    }

    fun clearJoinedRoom() {
        _uiState.value = _uiState.value.copy(joinedRoomId = null)
        _roomCode.value = ""
        pendingRoomCode = null
    }
}

data class JoinRoomUiState(
    val isLoading: Boolean = false,
    val error: String? = null,
    val showPasswordModal: Boolean = false,
    val isPasswordLoading: Boolean = false,
    val passwordError: String? = null,
    val joinedRoomId: Long? = null
)
```

### API Integration

**RoomRepository Extension**:

// Add to RoomRepository.kt
suspend fun joinRoom(roomCode: String, password: String?): Result<JoinRoomResponse> {
    return try {
        val response = api.joinRoom(JoinRoomRequest(
            roomCode = roomCode,
            password = password
        ))
        Result.success(response)
    } catch (e: HttpException) {
        when (e.code()) {
            401 -> Result.failure(PasswordIncorrectException())
            403 -> Result.failure(PasswordRequiredException())
            404 -> Result.failure(RoomNotFoundException())
            409 -> Result.failure(RoomFullException())
            else -> Result.failure(e)
        }
    } catch (e: Exception) {
        Result.failure(NetworkException(e))
    }
}

data class JoinRoomRequest(
    val roomCode: String,
    val password: String?
)

data class JoinRoomResponse(
    val roomId: Long,
    val permission: String
)

// Custom exceptions
class RoomNotFoundException : Exception("Room not found")
class PasswordRequiredException : Exception("Password required")
class PasswordIncorrectException : Exception("Incorrect password")
class RoomFullException : Exception("Room is full")
class NetworkException(cause: Throwable) : Exception("Network error", cause)
